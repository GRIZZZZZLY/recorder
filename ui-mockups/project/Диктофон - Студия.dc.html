@import "tailwindcss";

@theme {
  --color-canvas: #0d0e11;
  --color-surface: #16181d;
  --color-surface-elevated: #1e2127;
  --color-hairline: rgb(255 255 255 / 0.08);
  --color-hairline-strong: rgb(255 255 255 / 0.14);
  --color-text-primary: #f5f5f7;
  --color-text-secondary: #a1a1a6;
  --color-text-tertiary: #6e6e73;
  --color-accent: #ff453a;
  --color-accent-pressed: #d93b31;
  --color-ok: #30d158;
  --color-warn: #ffd60a;
  --color-waveform: #8e8e93;
  --color-focus: #f5f5f7;

  --radius-card: 12px;
  --radius-inner: 8px;
  --radius-chip: 999px;

  --duration-fast: 150ms;
  --duration-base: 240ms;
  --duration-slow: 420ms;
  --ease-out: cubic-bezier(0.2, 0, 0, 1);
  --ease-spring: cubic-bezier(0.34, 1.3, 0.64, 1);
  --stagger: 30ms;

  --font-sans: "Inter Tight", system-ui, sans-serif;
  --font-mono: "JetBrains Mono", ui-monospace, monospace;
}

@keyframes pulse-live {
  0%, 100% { opacity: 1; scale: 1; }
  50%      { opacity: 0.4; scale: 1.15; }
}
@keyframes shimmer {
  from { translate: -100% 0; }
  to   { translate: 100% 0; }
}
@keyframes cascade-in {
  from { opacity: 0; translate: 0 8px; }
  to   { opacity: 1; translate: 0 0; }
}

.live-dot   { animation: pulse-live 2s ease-in-out infinite; }
.shimmer::after {
  content: ""; position: absolute; inset: 0;
  background: linear-gradient(90deg, transparent, var(--color-hairline-strong), transparent);
  animation: shimmer 1.6s linear infinite;
}
.cascade > * {
  animation: cascade-in var(--duration-base) var(--ease-out) both;
  animation-delay: calc(min(var(--i), 12) * var(--stagger));
}

@media (prefers-reduced-motion: reduce) {
  .live-dot, .shimmer::after, .cascade > * { animation: none; }
  * { transition-duration: var(--duration-fast); transition-property: opacity; }
}
